add_library(canvas_runtime_control STATIC
  src/canvas_control_service.cpp
  src/build_info.cpp)
target_include_directories(canvas_runtime_control PUBLIC
  ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(canvas_runtime_control PUBLIC canvas_runtime_foundation)
target_compile_features(canvas_runtime_control PUBLIC cxx_std_20)
axiom_configure_build_info(canvas_runtime_control)
if(MSVC)
  target_compile_options(canvas_runtime_control PRIVATE /W4 /permissive- /WX /EHsc)
else()
  target_compile_options(canvas_runtime_control PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
