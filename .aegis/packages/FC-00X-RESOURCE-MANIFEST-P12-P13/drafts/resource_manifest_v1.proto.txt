message ResourceManifestEntry {
  Id128 resource_id = 1; // required, nonzero 16-byte Id128.value
  uint32 resource_kind = 2; // required: Image=1, Font=2, Brush=3
  uint64 resource_revision = 3; // required, 1..UINT64_MAX
  string content_hash = 4; // required, sha256:<64 lowercase hex>, 71 ASCII bytes
  uint64 byte_length = 5; // required, 1..UINT64_MAX; availability not required
  uint32 decode_version = 6; // required, 1
  uint32 layout_version = 7; // required, Font=1, Image/Brush=0, present zero
  bytes metadata = 8; // required, complete canonical kind-specific message above
}
message ResourceManifestTombstone {
  Id128 resource_id = 1; // required, matches preimage.resource_id
  uint64 retired_revision = 2; // required, preimage.resource_revision + 1
  string last_content_hash = 3; // required, equals preimage.content_hash
  Id128 removal_operation_id = 4; // required, OpId that performed removal
  ResourceManifestEntry preimage = 5; // required, full immutable last active entry
  bytes lineage_token = 6; // required, 32-byte derived SHA256 token
}
message ResourceManifest {
  uint32 schema_version = 1; // required, 1
  repeated ResourceManifestEntry entries = 2; // sorted complete ResourceId
  uint64 manifest_revision = 3; // required, starts 0, +1 per Applied manifest op
  repeated ResourceManifestTombstone retired_entries = 4; // sorted by ResourceId; same-ID coexistence allowed only when active revision > retired_revision
}
message ResourceExpectation {
  Id128 resource_id = 1; // required
  uint32 state = 2; // required, 0=never-used, 1=active, 2=retired
  uint64 revision = 3; // required, 0 for never, entry revision or retired_revision
}
message ReactivateResource {
  ResourceManifestEntry entry = 1; // required, current retired_revision + 1
  Id128 expected_removal_operation_id = 2; // required, exact tombstone OpId
  ResourceManifestEntry expected_preimage = 3; // required, exactly tombstone.preimage
  bytes expected_lineage_token = 4; // required, exactly tombstone.lineage_token, 32 bytes
}
message UpdateResourceManifestOp {
  repeated ResourceManifestEntry upserts = 1;
  repeated Id128 removes = 2;
  repeated ResourceExpectation expectations = 3;
  repeated ReactivateResource reactivations = 4;
  uint64 expected_manifest_revision = 5; // required, exact compare before prepare
}
