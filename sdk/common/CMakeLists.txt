add_library(axiom_sdk_session STATIC session.cpp)
target_link_libraries(axiom_sdk_session PUBLIC canvas_runtime_composition)
target_include_directories(axiom_sdk_session PUBLIC
  "${CMAKE_CURRENT_SOURCE_DIR}" "${PROJECT_SOURCE_DIR}/.deps/nlohmann-json/single_include")
target_compile_features(axiom_sdk_session PUBLIC cxx_std_20)
file(READ "${PROJECT_SOURCE_DIR}/sdk/version.json" _sdk_version_json)
string(JSON _sdk_version GET "${_sdk_version_json}" version)
target_compile_definitions(axiom_sdk_session PRIVATE AXIOM_SDK_VERSION="${_sdk_version}")
if(MSVC)
  target_compile_options(axiom_sdk_session PRIVATE /W4 /WX /permissive- /EHsc)
else()
  target_compile_options(axiom_sdk_session PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(EMSCRIPTEN)
    target_compile_options(axiom_sdk_session PRIVATE -fexceptions)
  endif()
endif()
