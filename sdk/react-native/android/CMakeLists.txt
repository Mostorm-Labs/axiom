cmake_minimum_required(VERSION 3.22)
project(AxiomRN LANGUAGES CXX)
if(NOT AXIOM_NATIVE_SDK_ROOT)
 message(FATAL_ERROR "AXIOM_NATIVE_SDK_ROOT must point to the matching Android native SDK")
endif()
add_library(axiom_runtime SHARED IMPORTED)
set_target_properties(axiom_runtime PROPERTIES IMPORTED_LOCATION "${AXIOM_NATIVE_SDK_ROOT}/lib/${ANDROID_ABI}/libaxiom.so" INTERFACE_INCLUDE_DIRECTORIES "${AXIOM_NATIVE_SDK_ROOT}/include")
add_library(axiom_react_native_android SHARED src/main/cpp/axiom_jni.cpp)
target_compile_features(axiom_react_native_android PRIVATE cxx_std_20)
target_link_libraries(axiom_react_native_android PRIVATE axiom_runtime android log)
