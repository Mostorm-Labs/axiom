cmake_minimum_required(VERSION 3.22)
project(AxiomRN LANGUAGES CXX)
if(NOT AXIOM_NATIVE_SDK_ROOT)
 message(FATAL_ERROR "AXIOM_NATIVE_SDK_ROOT must point to the matching Android native SDK")
endif()
add_library(axiom_native SHARED IMPORTED)
set_target_properties(axiom_native PROPERTIES IMPORTED_LOCATION "${AXIOM_NATIVE_SDK_ROOT}/lib/${ANDROID_ABI}/libaxiom_native.so" INTERFACE_INCLUDE_DIRECTORIES "${AXIOM_NATIVE_SDK_ROOT}/include")
add_library(axiom_rn SHARED src/main/cpp/axiom_jni.cpp)
target_compile_features(axiom_rn PRIVATE cxx_std_20)
target_link_libraries(axiom_rn PRIVATE axiom_native android log)
