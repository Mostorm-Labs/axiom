cmake_minimum_required(VERSION 3.30)
project(AxiomDllConsumer LANGUAGES C CXX)
find_package(AxiomSDK 1 CONFIG REQUIRED)
add_executable(axiom_example main.cpp)
target_link_libraries(axiom_example PRIVATE Axiom::SDK comdlg32)
target_compile_features(axiom_example PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(axiom_example PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
  set_property(TARGET axiom_example PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
endif()
axiom_deploy(axiom_example)
foreach(ext png jpg webp)
  configure_file(fixtures/sample.${ext} sample.${ext} COPYONLY)
endforeach()
enable_testing()
add_test(NAME ExternalAxiomDll COMMAND axiom_example --test)
set_tests_properties(ExternalAxiomDll PROPERTIES TIMEOUT 90)
add_executable(axiom_c_loader load_library.c)
target_include_directories(axiom_c_loader PRIVATE "$<TARGET_PROPERTY:Axiom::SDK,INTERFACE_INCLUDE_DIRECTORIES>")
if(MSVC)
  target_compile_options(axiom_c_loader PRIVATE /W4 /WX)
  set_property(TARGET axiom_c_loader PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
endif()
axiom_deploy(axiom_c_loader)
add_test(NAME ExternalAxiomLoadLibrary COMMAND axiom_c_loader)
set_tests_properties(ExternalAxiomLoadLibrary PROPERTIES TIMEOUT 90)
