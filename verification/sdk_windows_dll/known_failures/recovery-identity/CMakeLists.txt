cmake_minimum_required(VERSION 3.30)
project(AxiomRecoveryIdentityProbe LANGUAGES CXX)
find_package(AxiomSDK 1 CONFIG REQUIRED)
add_executable(axiom_recovery_identity_probe main.cpp)
target_link_libraries(axiom_recovery_identity_probe PRIVATE Axiom::SDK)
target_compile_features(axiom_recovery_identity_probe PRIVATE cxx_std_20)
target_compile_options(axiom_recovery_identity_probe PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
set_property(TARGET axiom_recovery_identity_probe PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
axiom_deploy(axiom_recovery_identity_probe)
enable_testing()
add_test(NAME SDKRecoveryOperationIdentity COMMAND axiom_recovery_identity_probe)
set_tests_properties(SDKRecoveryOperationIdentity PROPERTIES TIMEOUT 90)
