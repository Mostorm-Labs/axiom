cmake_minimum_required(VERSION 3.22)
project(axiom_sdk_external_consumer LANGUAGES C CXX)
enable_testing()

set(AXIOM_SDK_INCLUDE_DIR "" CACHE PATH "Directory containing the installed canonical V1 header")
set(AXIOM_SDK_LIBRARY "" CACHE FILEPATH "Installed SDK library or Windows import library")
option(AXIOM_SDK_SHARED "Use the SDK as a shared library" OFF)
if(NOT EXISTS "${AXIOM_SDK_INCLUDE_DIR}/canvas_runtime_api_v1.h")
  message(FATAL_ERROR "Installed V1 header is missing; set AXIOM_SDK_INCLUDE_DIR")
endif()
if(NOT EXISTS "${AXIOM_SDK_LIBRARY}" OR IS_DIRECTORY "${AXIOM_SDK_LIBRARY}")
  message(FATAL_ERROR "Installed SDK library is missing; set AXIOM_SDK_LIBRARY")
endif()

add_executable(sdk01_external_c_smoke external_smoke.c)
add_executable(sdk01_external_cpp_link external_link.cpp)
foreach(target sdk01_external_c_smoke sdk01_external_cpp_link)
  target_include_directories(${target} PRIVATE "${AXIOM_SDK_INCLUDE_DIR}")
  target_link_libraries(${target} PRIVATE "${AXIOM_SDK_LIBRARY}")
  if(AXIOM_SDK_SHARED)
    target_compile_definitions(${target} PRIVATE CANVAS_RUNTIME_SHARED_LIBRARY)
  endif()
endforeach()
target_compile_features(sdk01_external_c_smoke PRIVATE c_std_11)
target_compile_features(sdk01_external_cpp_link PRIVATE cxx_std_17)
add_test(NAME SDK01ExternalCSmoke COMMAND sdk01_external_c_smoke)
add_test(NAME SDK01ExternalCppLink COMMAND sdk01_external_cpp_link)
