cmake_minimum_required(VERSION 3.24)
project(axiom_external_consumer LANGUAGES CXX)
if(AXIOM_TEST_BAD_CRT)
  set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
else()
  set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
endif()
find_package(AxiomRuntime CONFIG REQUIRED)
add_executable(external_consumer main.cpp)
target_link_libraries(external_consumer PRIVATE Axiom::Runtime)
target_compile_features(external_consumer PRIVATE cxx_std_20)
if(MSVC)
  set_property(TARGET external_consumer PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
  target_compile_options(external_consumer PRIVATE /W4 /WX /permissive- /EHsc)
endif()
enable_testing()
add_test(NAME ExternalRuntimeLifecycle COMMAND external_consumer)
