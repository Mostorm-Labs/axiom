  if(NOT case STREQUAL "incremental_runtime_atomic_publication")
    target_sources(g5_${case} PRIVATE "${AXIOM_G5_SOURCE}/runtime/scene/tests/incremental_runtime_full_oracle_adapter.cpp")
  endif()
  target_include_directories(g5_${case} PRIVATE "${AXIOM_G5_SOURCE}/runtime/semantic/src")
  target_link_libraries(g5_${case} PRIVATE g5_scene_fakes)
  target_compile_features(g5_${case} PRIVATE cxx_std_20)
  add_test(NAME G5_${case} COMMAND g5_${case})
