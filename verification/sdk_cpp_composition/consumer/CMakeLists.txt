cmake_minimum_required(VERSION 3.24)
project(axiom_public_composition LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
find_package(AxiomRuntime CONFIG REQUIRED)
add_executable(single_owner single_owner.cpp)
target_link_libraries(single_owner PRIVATE Axiom::Runtime)
target_compile_features(single_owner PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(single_owner PRIVATE /W4 /WX /permissive- /EHsc /showIncludes)
endif()
axiom_runtime_copy_assets(single_owner)
enable_testing()
add_test(NAME ExternalCompositionSingleOwner COMMAND single_owner)
